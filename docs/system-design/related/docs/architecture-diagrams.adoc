== Flow Charts

=== Boot / `setup()` flow

[plantuml, target=flow-boot]
....
@startuml
start
:Read device_id from NVS
Derive device name from MAC;
if (Pairing document on LittleFS?) then (no)
  :RunProvisioningMode
  (BLE provisioning, see sequence diagram);
  :Re-read pairing document;
else (yes)
endif
:CloudService.SetConfig + Log.configure
(base URL + API key from pairing doc);
:WiFi: SetCredentials + EnsureConnected;
if (WiFi connected?) then (yes)
  :NTP sync, set start time,
  UpdateLastSeen, ReportModelVersion;
else (no)
endif
if (Plant label known?) then (no)
  :FetchPlantLabel from cloud;
  if (Label found or just provisioned?) then (no)
    :PerformFactoryReset
    (clear history, unpair, delete device,
    clear NVS, restart); <<#pink>>
    stop
  else (yes)
  endif
else (yes)
endif
:FetchProfileSettings → SetPlantProfile;
:LoadHistoryFile from LittleFS →
LoadHistoricalSnapshots;
if (MonitoringSystem.Init OK?) then (no)
  :Halt forever; <<#pink>>
  stop
else (yes)
endif
if (Plant label set?) then (yes)
  :Run first monitoring cycle
  + UpdateLastSeen;
else (no)
  :Wait for plant setup in app;
endif
:Log.uploadPending;
:enter loop();
stop
@enduml

=== Main `loop()` flow

[plantuml, target=flow-loop]
....
@startuml
start
:loop tick;
if (5 s since last command check?) then (yes)
  :Reconnect WiFi if down;
  if (Plant label empty and device_id set?) then (yes)
    if (FetchPlantLabel succeeded?) then (yes)
      :Fetch profile, set it,
      run first cycle, UpdateLastSeen,
      reset no-plant counter;
    else (no)
      if (No-plant counter >= 30 ticks?) then (yes)
        :PerformFactoryReset; <<#pink>>
        stop
      else (no)
        :Log grace-period count;
      endif
    endif
  else (no)
    :Reset no-plant counter;
  endif
  if (trigger_reset set in cloud?) then (yes)
    :PerformFactoryReset; <<#pink>>
    stop
  else (no)
  endif
  if (trigger_measurement set in cloud?) then (yes)
    :RunMonitoringCycle + UpdateLastSeen;
  else (no)
  endif
  :Log.uploadPending;
else (no)
endif
if (3 h since last scheduled cycle?) then (yes)
  :Reconnect WiFi if down;
  :FetchProfileSettings → SetPlantProfile;
  :RunMonitoringCycle + UpdateLastSeen;
else (no)
endif
stop
@enduml

:FeatureEngineering::Build → 13-feature vector;

  :MLLayer::Predict →

participant "ProvisioningService\n(BLE)" as Prov
participant "NvsStorage" as NVS
participant "WiFiCommunication" as WiFi
participant "TimeService" as Time
participant "CloudService" as Cloud
participant "FileStorageService" as FS

Main -> Prov : getOrCreateDeviceId()
Prov -> NVS : read/write "device_id"
NVS --> Prov : device UUID
Main -> Prov : begin(deviceName, deviceId)
note over Prov : Advertise PHM service\nUUID 4fafc201-…914b

App -> Prov : connect (BLE central)
App -> Prov : read DeviceId characteristic
Prov --> App : device UUID
App -> Prov : write SSID
App -> Prov : write Password
App -> Prov : write API key
App -> Prov : write Supabase URL

loop until isProvisioned()
  Main -> Prov : maintainAdvertising()
  note over Prov : re-advertise if app\ndisconnected mid-flow
end

Main -> Prov : stop()
Main -> WiFi : SetCredentials(ssid, pass)
Main -> WiFi : EnsureConnected()
alt WiFi failed
  Main -> NVS : clearAll()
  Main -> Main : ESP.restart()
else WiFi connected
  Main -> Cloud : SetConfig(base + /rest/v1, apiKey)
  Main -> Time : SyncTimeWithNtp(10 s)
  Main -> Cloud : RegisterDevice(id, name, now)
  Main -> NVS : writeString("device_id", id)
  Main -> FS : WritePaired(ssid, pass, apiKey, url)
  note over Main : gJustProvisioned = true
end
@enduml

=== Monitoring cycle + cloud upload

[plantuml, target=seq-cycle]
....
@startuml
autonumber
participant "main.cpp" as Main
participant "MonitoringSystem" as Mon
participant "TempHumiditySensor" as DHT
participant "SoilMoistureSensor" as Soil
participant "LightSensor" as Light
participant "SensorHistory" as Hist
participant "FeatureEngineering" as FE
participant "MLLayer" as ML
participant "PredictiveIrrigationModel" as Irr
participant "RecommendationEngine" as Rec
participant "FileStorageService" as FS
participant "CloudService" as Cloud

Main -> Mon : RunCycleDetailed()
activate Mon
Mon -> DHT : Read()
DHT --> Mon : (temperature, humidity)
Mon -> Soil : ReadPercent()
Soil --> Mon : soil moisture %
Mon -> Light : ReadLightLevelPct()
Light --> Mon : light %
Mon -> Hist : Add(snapshot)
Mon -> FE : Build(history)
FE --> Mon : FeatureVector (13)
Mon -> ML : Predict(features)
ML --> Mon : MLResult (risk, confidence)
Mon -> Irr : PredictMinutesUntilWatering(features, profile)
Irr --> Mon : minutes
Mon -> Rec : Build(snapshot, mlResult, profile, minutes)
Rec --> Mon : Recommendation
Mon --> Main : MonitoringCycleResult
deactivate Mon

Main -> FS : AppendToHistoryFile(result, label, id)
alt plant label set
  Main -> Cloud : SendReading(json) → plant_readings
  Main -> Cloud : SendModelMetrics(json) → model_metrics
else no plant label
  Main -> Main : skip upload
end

=== Cloud command polling (every 5 s)

[plantuml, target=seq-polling]
....
@startuml
autonumber
participant "main.cpp (loop)" as Main
participant "WiFiCommunication" as WiFi
participant "CloudService" as Cloud
participant "Supabase REST" as SB
participant "CloudLogger" as Log

Main -> WiFi : EnsureConnected() (if down)
Main -> Cloud : CheckTriggerReset(deviceId)
Cloud -> SB : GET devices?trigger_reset
SB --> Cloud : flag
alt trigger_reset set
  Cloud --> Main : true
  Main -> Main : PerformFactoryReset()
else not set
  Cloud --> Main : false
  Main -> Cloud : CheckTriggerMeasurement(deviceId)
  Cloud -> SB : GET + clear trigger_measurement
  alt trigger_measurement set
    Cloud --> Main : true
    Main -> Main : RunMonitoringCycle()
    Main -> Cloud : UpdateLastSeen(deviceId, now)
  else not set
    Cloud --> Main : false
  end
end
Main -> Log : uploadPending(deviceId, deviceName)
Log -> SB : batched insert into logs\n(if queued + online)
@enduml
